 DHCP IM NETZ 50 REPARIEREN
---------------------------------------------------------------------
# Eingrenzen von aussen nach innen: Client -> Relay -> Server

[PC2]
ip -4 addr show ens3                 # keine IPv4-Adresse
sudo networkctl reconfigure ens3     # neue Anfrage auslösen

[R1]
sudo tcpdump -ni eth3 -c 6 port 67 or port 68
# DISCOVER von PC2 kommen an -> Client und Kabel sind in Ordnung

[SRV-INFRA]
sudo tcpdump -vni ens3 -c 6 port 67
# DISCOVER kommen vom Relay 10.10.30.1 an, Gateway-IP (giaddr) 10.10.50.1
# Es geht KEIN OFFER zurück -> Problem liegt beim DHCP-Server

journalctl -u kea-dhcp4-server -n 40 --no-pager
# Befund 1: "No subnet matches relay address: 10.10.50.1"
#           "failed to select a subnet for incoming packet"

cat /etc/kea/kea-dhcp4.conf
# Befund 1: "subnet": "10.10.5.0/24" und Pool 10.10.5.100 - 10.10.5.199
#           -> passt nicht zum Netz 10.10.50.0/24 (ANF-NET-01)
# Befund 2: "domain-name-servers": "10.10.30.53"
#           -> verlangt ist 10.10.50.1 (ANF-NET-02)
#           Dieser Fehler fällt erst auf, wenn PC2 eine Adresse hat.

# Rückfallpunkt setzen, dann ändern
sudo cp /etc/kea/kea-dhcp4.conf /etc/kea/kea-dhcp4.conf.bak
sudo nano /etc/kea/kea-dhcp4.conf

# Korrekte Werte:
#   "data": "10.10.50.1"                          (domain-name-servers)
#   "subnet": "10.10.50.0/24",
#   "pools": [ { "pool": "10.10.50.100 - 10.10.50.199" } ],

sudo kea-dhcp4 -t /etc/kea/kea-dhcp4.conf      # Syntax prüfen
sudo systemctl restart kea-dhcp4-server
diff /etc/kea/kea-dhcp4.conf.bak /etc/kea/kea-dhcp4.conf

 REVERSE PROXY äNDERN (CHANGE «SHOP NUR üBER HTTPS»)
---------------------------------------------------------------------
# Reihenfolge: Proxy konfigurieren und lokal prüfen -> Firewalls ->
# Test über IP -> Namen umstellen -> Test über Namen
# Rückfallpunkte: shop.bak auf SRV-PROXY, alte DNS-Adresse 10.10.40.14

[SRV-PROXY]
sudo cp /etc/nginx/sites-available/shop /etc/nginx/sites-available/shop.bak
sudo nano /etc/nginx/sites-available/shop

# ---- Inhalt /etc/nginx/sites-available/shop (Lösung) ----
upstream shop_backend {
    server 10.10.40.14:8080 weight=2;   # shop-a, aktuelle Version
    server 10.10.40.14:8081 weight=1;   # shop-b, neue Version (Canary)
}

server {
    listen 80;
    server_name shop.techtrade.lab;
    return 301 https://$host$request_uri;   # dauerhafte Umleitung
}

server {
    listen 443 ssl;
    server_name shop.techtrade.lab;

    ssl_certificate     /etc/nginx/ssl/shop.crt;
    ssl_certificate_key /etc/nginx/ssl/shop.key;

    location / {
        proxy_pass http://shop_backend;
        proxy_set_header Host              $host;
        proxy_set_header X-Real-IP         $remote_addr;
        proxy_set_header X-Forwarded-For   $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
}
# ---- Ende ----

sudo nginx -t
sudo systemctl reload nginx
sudo ss -tlnp | grep nginx                       # :80 und :443
curl -sk -D - -o /dev/null https://127.0.0.1/health

[R1]   Firewall 1: Clients -> SRV-PROXY 80/443, kein direkter Zugriff auf 8080
configure
set firewall group address-group SRV-PROXY address '10.10.40.10'
set firewall ipv4 forward filter rule 21 action 'accept'
set firewall ipv4 forward filter rule 21 description 'Clients -> Reverse Proxy'
set firewall ipv4 forward filter rule 21 protocol 'tcp'
set firewall ipv4 forward filter rule 21 source group network-group 'NET-CLIENTS'
set firewall ipv4 forward filter rule 21 destination group address-group 'SRV-PROXY'
set firewall ipv4 forward filter rule 21 destination port '80,443'
delete firewall group port-group PORTS-SHOP port '8080'
commit-confirm 5
confirm
save
exit
